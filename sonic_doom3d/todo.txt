int   mapWidth()  { return static_cast<int>(kMap.front().size()); }

float clampf(float v, float lo, float hi) { return std::max(lo, std::min(hi, v)); }
float length(const Vec2& v)               { return std::sqrt(v.x*v.x + v.y*v.y); }
float lengthSq(const Vec2& v)             { return v.x*v.x + v.y*v.y; }

Vec2 normalizeOrZero(const Vec2& v) {
    float len = length(v);
    if (len < 0.0001f) return {0,0};
    return {v.x/len, v.y/len};
}
Vec2 add(const Vec2& a, const Vec2& b) { return {a.x+b.x, a.y+b.y}; }
Vec2 sub(const Vec2& a, const Vec2& b) { return {a.x-b.x, a.y-b.y}; }
Vec2 mul(const Vec2& v, float s)       { return {v.x*s, v.y*s}; }

Vec2 moveTowards(const Vec2& cur, const Vec2& tgt, float maxDelta) {
    Vec2 d = sub(tgt, cur);
    float dist = length(d);
    if (dist <= maxDelta || dist < 0.0001f) return tgt;
    return add(cur, mul(d, maxDelta/dist));
}

float randRange(float lo, float hi) {
    std::uniform_real_distribution<float> dist(lo, hi);
    return dist(gRng);
}

char mapCell(int x, int y) {
    if (x < 0 || y < 0 || y >= mapHeight() || x >= mapWidth()) return '#';
    return kMap[y][x];
}

bool isWall(int x, int y) {
    if (x < 0 || y < 0 || y >= mapHeight() || x >= mapWidth()) return true;
    return kMap[y][x] == '#';
}

bool isWallF(float x, float y) {
    return isWall(static_cast<int>(x), static_cast<int>(y));
}

// Generates procedural Daggerfall-esque dungeon textures
void generateTextures() {
    for(int y = 0; y < kTexSize; ++y) {
        for(int x = 0; x < kTexSize; ++x) {
            // Dark stone bricks
            bool edgeX = (x % 32 == 0);
            bool edgeY = (y % 16 == 0);
            int stagger = (y / 16) % 2 == 0 ? 0 : 16;
            bool mortar = edgeY || ((x + stagger) % 32 == 0);
            
            int noise = rand() % 40;
            if (mortar) {
                gWallTex[y * kTexSize + x] = (20 << 16) | (20 << 8) | 20; // Dark grey mortar
            } else {
                int col = 60 + noise; // Stone color
                gWallTex[y * kTexSize + x] = (col << 16) | (col << 8) | (col + 10);
            }

            // Mossy cobble floor
            int fnoise = rand() % 50;
            int r = 30 + fnoise/2;
            int g = 40 + fnoise; // greener
            int b = 30 + fnoise/2;
            gFloorTex[y * kTexSize + x] = (r << 16) | (g << 8) | b;
        }
    }
}

// ============================================================
//  PIXEL OUTPUT
// ============================================================
inline void putPixel(int x, int y, unsigned int c) {
    if ((unsigned)x < (unsigned)kScreenWidth && (unsigned)y < (unsigned)kScreenHeight)
        gPixels[y * kScreenWidth + x] = c;
}

void fillRect(int x, int y, int w, int h, unsigned int c) {
    int x0=std::max(0,x), y0=std::max(0,y);
    int x1=std::min(kScreenWidth,x+w), y1=std::min(kScreenHeight,y+h);
    for (int py=y0; py<y1; ++py) {
        unsigned int* row = &gPixels[py*kScreenWidth];
        for (int px=x0; px<x1; ++px) row[px]=c;
    }
}

void clearScreen(unsigned int c) { std::fill(gPixels.begin(), gPixels.end(), c); }

void drawCircleOutline(int cx, int cy, int r, int thick, unsigned int c) {
    int o2=(r)*(r), i2=std::max(0,r-thick); i2*=i2;
    for (int dy=-r; dy<=r; ++dy)
        for (int dx=-r; dx<=r; ++dx) {
            int d2=dx*dx+dy*dy;
            if (d2<=o2 && d2>=i2) putPixel(cx+dx,cy+dy,c);
        }
}

void drawCircleFilled(int cx, int cy, int r, unsigned int c) {
    int r2=r*r;
    for (int dy=-r; dy<=r; ++dy)
        for (int dx=-r; dx<=r; ++dx)
            if (dx*dx+dy*dy<=r2) putPixel(cx+dx,cy+dy,c);
}

void drawLine(int x0, int y0, int x1, int y1, unsigned int c) {
    int dx = std::abs(x1 - x0), sx = x0 < x1 ? 1 : -1;
    int dy = -std::abs(y1 - y0), sy = y0 < y1 ? 1 : -1;
    int err = dx + dy;
    for (;;) {
        putPixel(x0, y0, c);
        if (x0 == x1 && y0 == y1) break;
        int e2 = err * 2;
        if (e2 >= dy) { err += dy; x0 += sx; }
        if (e2 <= dx) { err += dx; y0 += sy; }
    }
}

void initRayLut() {
    for (int x=0; x<kScreenWidth; ++x)
        gCameraX[x] = 2.0f*x/(float)kScreenWidth - 1.0f;
}

// ============================================================
//  BACK-BUFFER
// ============================================================
void createBackBuffer(HWND hWnd) {
    HDC hdc = GetDC(hWnd);
    gBackDC  = CreateCompatibleDC(hdc);

    BITMAPINFO bmi = {};
    bmi.bmiHeader.biSize        = sizeof(BITMAPINFOHEADER);
    bmi.bmiHeader.biWidth       = kScreenWidth;
    bmi.bmiHeader.biHeight      = -kScreenHeight;
    bmi.bmiHeader.biPlanes      = 1;
    bmi.bmiHeader.biBitCount    = 32;
    bmi.bmiHeader.biCompression = BI_RGB;

    gBackBmp = CreateDIBSection(gBackDC, &bmi, DIB_RGB_COLORS, &gBackBits, NULL, 0);
    SelectObject(gBackDC, gBackBmp);
    ReleaseDC(hWnd, hdc);
}

void presentFrame(HWND hWnd) {
    std::memcpy(gBackBits, gPixels.data(), kScreenWidth * kScreenHeight * 4);
    HDC hdc = GetDC(hWnd);
    BitBlt(hdc, 0, 0, kScreenWidth, kScreenHeight, gBackDC, 0, 0, SRCCOPY);
    ReleaseDC(hWnd, hdc);
}

void resolveWallCollision(Player& player) {
    const float r = kPlayerRadius;
    if (isWallF(player.pos.x + r, player.pos.y) || isWallF(player.pos.x - r, player.pos.y)) {
        int tx = static_cast<int>(player.pos.x + (player.vel.x > 0 ? r : -r));
        float cx = tx + (player.vel.x > 0 ? 0.0f : 1.0f);
        player.pos.x = cx + (player.vel.x > 0 ? -r - 0.001f : r + 0.001f);
        player.vel.x *= -0.3f;
    }
    if (isWallF(player.pos.x, player.pos.y + r) || isWallF(player.pos.x, player.pos.y - r)) {
        int ty = static_cast<int>(player.pos.y + (player.vel.y > 0 ? r : -r));
        float cy = ty + (player.vel.y > 0 ? 0.0f : 1.0f);
        player.pos.y = cy + (player.vel.y > 0 ? -r - 0.001f : r + 0.001f);
        player.vel.y *= -0.3f;
    }
    static const std::array<Vec2,4> corners = { Vec2{r, r}, Vec2{-r, r}, Vec2{r, -r}, Vec2{-r, -r} };
    for (auto& c : corners) {
        int cx = static_cast<int>(player.pos.x + c.x), cy = static_cast<int>(player.pos.y + c.y);
        if (isWall(cx, cy)) {
            Vec2 away = normalizeOrZero({player.pos.x - (cx+0.5f), player.pos.y - (cy+0.5f)});
            if (length(away) < 0.0001f) away = {1,0};
            player.pos = add(player.pos, mul(away, 0.04f));
            player.vel = mul(player.vel, 0.75f);
        }
    }
}

// ============================================================
//  RAYCASTING (UPDATED FOR TEXTURES)
// ============================================================
RayHit raycastDistance(const Player& player, float rdx, float rdy) {
    int mx = static_cast<int>(player.pos.x);
    int my = static_cast<int>(player.pos.y);

    float ddx = (std::fabs(rdx) < 0.0001f) ? 1e6f : std::fabs(1.0f/rdx);
    float ddy = (std::fabs(rdy) < 0.0001f) ? 1e6f : std::fabs(1.0f/rdy);

    int sx = rdx < 0 ? -1 : 1;
    int sy = rdy < 0 ? -1 : 1;
    float sdx = rdx < 0 ? (player.pos.x - mx)*ddx : (mx+1.0f - player.pos.x)*ddx;
    float sdy = rdy < 0 ? (player.pos.y - my)*ddy : (my+1.0f - player.pos.y)*ddy;

    bool side = false;
    for (;;) {
        if (sdx < sdy) { sdx+=ddx; mx+=sx; side=false; }
        else           { sdy+=ddy; my+=sy; side=true;  }
        if (isWall(mx,my)) break;
        if (std::min(sdx,sdy) > kMaxRayDistance) return {kMaxRayDistance, side, 0.0f, mx, my};
    }

    float dist;
    float wallX;
    if (!side) {
        dist = (mx - player.pos.x + (1 - sx) * 0.5f) / rdx;
        wallX = player.pos.y + dist * rdy;
    } else {
        dist = (my - player.pos.y + (1 - sy) * 0.5f) / rdy;
        wallX = player.pos.x + dist * rdx;
    }
    wallX -= std::floor(wallX);
    return {clampf(std::fabs(dist), 0.001f, kMaxRayDistance), side, wallX, mx, my};
}

// ============================================================
//  BILLBOARDS & HUD
// ============================================================
// (Billboard code remains functionally identical, omitted visually changing for brevity, 
//  but you can apply lighting depth to these colors too!)
void drawSnowBillboards(const Player& player, const std::vector<SnowParticle>& snow, float horizon) {
    for (auto& p : snow) {
        Vec2 d = sub(p.pos, player.pos);
        float dist = length(d);
        if (dist < 0.1f || dist > 18.0f) continue;
        float a = std::atan2(d.y, d.x) - player.angle;
        while (a > 3.14159f) a -= 6.28318f;
        while (a < -3.14159f) a += 6.28318f;
        if (std::fabs(a) > kHalfFov) continue;
        int sx = (int)((a/kHalfFov*0.5f+0.5f)*(float)kScreenWidth);
        int cy = (int)(kScreenHeight*0.5f + horizon - player.z*130.0f - p.z*120.0f);
        int sz = (int)clampf(56.0f/dist, 1.0f, 3.0f);
        
        // Depth shading for snow
        float shade = clampf(1.0f - (dist / 16.0f), 0.0f, 1.0f);
        int c = (int)(255 * shade);
        unsigned int color = (c << 16) | (c << 8) | c;
        
        drawCircleFilled(sx, cy, sz, color);
    }
}

// ============================================================
//  WINMAIN
// ============================================================
int APIENTRY WinMain(HINSTANCE hInstance, HINSTANCE, LPSTR, int nCmdShow) {
    WNDCLASSA wc = {};
    wc.lpfnWndProc   = WndProc;
    wc.hInstance     = hInstance;
    wc.lpszClassName = "DaggerfallRenderer";
    wc.hCursor       = LoadCursor(NULL, IDC_ARROW);
    RegisterClassA(&wc);

    HWND hWnd = CreateWindowA("DaggerfallRenderer", "Daggerfall Engine Clone",
        WS_OVERLAPPEDWINDOW & ~WS_SIZEBOX, CW_USEDEFAULT, CW_USEDEFAULT,
        kScreenWidth+16, kScreenHeight+39, NULL, NULL, hInstance, NULL);
    if (!hWnd) return 0;

    ShowWindow(hWnd, nCmdShow);
    createBackBuffer(hWnd);
    initRayLut();
    generateTextures(); // Generate procedural retro textures

    Player player = {{2.5f, 2.5f}, {0,0}, 0, 0, 0, true};
    std::vector<Ring> rings = gatherRings();
    std::vector<SnowParticle> snow = createSnow(player);
    float levelTimer = 0.0f;
    int collectedRings = 0;

    LARGE_INTEGER freq, last;
    QueryPerformanceFrequency(&freq);
    QueryPerformanceCounter(&last);
    const float kFrameTime = 1.0f / (float)kTargetFPS;
    MSG msg = {};
    bool running = true;

    while (running) {
        while (PeekMessage(&msg, NULL, 0, 0, PM_REMOVE)) {
            if (msg.message == WM_QUIT) running = false;
            TranslateMessage(&msg); DispatchMessage(&msg);
        }

        LARGE_INTEGER now;
        QueryPerformanceCounter(&now);
        float dt = (float)(now.QuadPart - last.QuadPart) / (float)freq.QuadPart;
        if (dt < kFrameTime) { Sleep(1); continue; }
        last = now;
        dt = clampf(dt, 0.0f, 0.033f);
        levelTimer += dt;

        // Turn & Move
        float turn = 0;
        if (GetAsyncKeyState(VK_LEFT)  & 0x8000) turn -= 1;
        if (GetAsyncKeyState(VK_RIGHT) & 0x8000) turn += 1;
        player.angle += turn * kTurnRate * dt;

        Vec2 fwd = {std::cos(player.angle), std::sin(player.angle)};
        Vec2 rgt = {-fwd.y, fwd.x};
        Vec2 mi = {0,0};
        if (GetAsyncKeyState('W') & 0x8000) mi = add(mi, fwd);
        if (GetAsyncKeyState('S') & 0x8000) mi = sub(mi, fwd);
        if (GetAsyncKeyState('D') & 0x8000) mi = add(mi, rgt);
        if (GetAsyncKeyState('A') & 0x8000) mi = sub(mi, rgt);
        mi = normalizeOrZero(mi);
        
        player.vel = moveTowards(player.vel, mul(mi, kTopRunSpeed), kAccelGround*dt);
        player.vel = mul(player.vel, std::max(0.0f, 1.0f - kGroundFriction*dt));
        player.pos = add(player.pos, mul(player.vel, dt));
        resolveWallCollision(player);

        // Render Prep
        float speed = length(player.vel);
        float bob = std::sin(levelTimer*(6.0f+speed*0.7f)) * clampf(speed/24.0f,0,1) * 16.0f;
        float horizon = bob - player.z*120.0f;
        Vec2 plane = {-fwd.y*kCameraPlaneScale, fwd.x*kCameraPlaneScale};

        clearScreen(0x00000000); // Pitch black background (Daggerfall voids)

        // ----------------------------------------------------
        // THE NEW DAGGERFALL RENDERING LOOP
        // ----------------------------------------------------
        for (int x=0; x<kScreenWidth; ++x) {
            float rdx = fwd.x + plane.x*gCameraX[x];
            float rdy = fwd.y + plane.y*gCameraX[x];
            
            RayHit hit = raycastDistance(player, rdx, rdy);
            
            float lh = (float)kScreenHeight / hit.dist;
            int ds = std::max(0, (int)(-lh/2 + kScreenHeight/2 + horizon));
            int de = std::min(kScreenHeight, (int)(lh/2 + kScreenHeight/2 + horizon));

            // 1. Texturing the walls
            int texX = (int)(hit.wallX * kTexSize);
            if((!hit.side && rdx > 0) || (hit.side && rdy < 0)) texX = kTexSize - texX - 1;
            
            float step = 1.0f * kTexSize / lh;
            float texPos = (ds - horizon - kScreenHeight/2 + lh/2) * step;

            // Fog multiplier (distance-based darkness)
            float fogDist = clampf(1.0f - (hit.dist / 14.0f), 0.0f, 1.0f);
            float shadow = hit.side ? 0.65f : 1.0f; // Fake directional lighting
            float shade = fogDist * shadow;

            for(int y = ds; y < de; y++) {
                int texY = (int)texPos & (kTexSize - 1);
                texPos += step;
                unsigned int color = gWallTex[texY * kTexSize + texX];
                
                int r = (int)(((color >> 16) & 0xFF) * shade);
                int g = (int)(((color >> 8) & 0xFF) * shade);
                int b = (int)((color & 0xFF) * shade);
                putPixel(x, y, (r << 16) | (g << 8) | b);
            }

            // 2. Floor Casting
            // To emulate Daggerfall's textured floors beneath the player
            for(int y = de; y < kScreenHeight; y++) {
                float p = y - kScreenHeight/2.0f - horizon;
                if (p <= 0) continue; // safety
                
                // Magic formula for floor casting with height/pitch
                float camZ = 0.5f * kScreenHeight + player.z * 130.0f;
                float rowDistance = camZ / p;
                
                // Note: The cosine of the ray angle is built into the ray direction, 
                // but we need to divide by the view plane distance to fix fisheye on the floor
                float fixDist = rowDistance / std::cos(std::atan2(gCameraX[x] * kCameraPlaneScale, 1.0f));

                float floorX = player.pos.x + rdx * fixDist;
                float floorY = player.pos.y + rdy * fixDist;
                
                int tx = (int)(floorX * kTexSize) & (kTexSize - 1);
                int ty = (int)(floorY * kTexSize) & (kTexSize - 1);
                
                unsigned int color = gFloorTex[ty * kTexSize + tx];
                
                // Fog for floor
                float floorFog = clampf(1.0f - (fixDist / 14.0f), 0.0f, 1.0f);
                int r = (int)(((color >> 16) & 0xFF) * floorFog);
                int g = (int)(((color >> 8) & 0xFF) * floorFog);
                int b = (int)((color & 0xFF) * floorFog);
                putPixel(x, y, (r << 16) | (g << 8) | b);
            }
        }

        // Draw HUD & Blit
        presentFrame(hWnd);
        HDC hdc = GetDC(hWnd);
        drawHud(hdc, collectedRings, 10, (int)(speed*10), levelTimer, false, false, 3, 0, false);
        ReleaseDC(hWnd, hdc);
    }

    if (gBackBmp) DeleteObject(gBackBmp);
    if (gBackDC)  DeleteDC(gBackDC);
    return 0;
}
